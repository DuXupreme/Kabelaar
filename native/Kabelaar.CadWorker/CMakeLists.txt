cmake_minimum_required(VERSION 3.24)
project(KabelaarCadWorker VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Link only geometry/data exchange: no Draw, OpenGL, VTK or Python runtime.
set(OCCT_ROOT "" CACHE PATH "Official Open CASCADE 8.0.1 SDK directory")
find_path(OCCT_INCLUDE_DIR STEPControl_Reader.hxx PATHS "${OCCT_ROOT}/inc" REQUIRED NO_DEFAULT_PATH)
add_executable(Kabelaar.CadWorker main.cpp)
target_include_directories(Kabelaar.CadWorker PRIVATE "${OCCT_INCLUDE_DIR}")
foreach(lib TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKMesh TKHLR TKXSBase TKDESTEP TKPrim TKBO TKBool)
  find_library(OCCT_${lib} NAMES ${lib} PATHS "${OCCT_ROOT}/win64/vc14/lib" REQUIRED NO_DEFAULT_PATH)
  target_link_libraries(Kabelaar.CadWorker PRIVATE "${OCCT_${lib}}")
endforeach()
if(MSVC)
  target_compile_options(Kabelaar.CadWorker PRIVATE /EHsc /utf-8 /W4)
endif()
install(TARGETS Kabelaar.CadWorker RUNTIME DESTINATION .)
